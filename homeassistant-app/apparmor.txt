#include <tunables/global>

# Enforced least-privilege profile refined through Home Assistant acceptance
# testing. Access outside this allowlist is denied and recorded by AppArmor.
profile layerv_ha_gateway flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # Packaged Python runtime, Gateway sources and fixed child executables.
  /usr/local/bin/python rix,
  /usr/local/bin/python3 rix,
  /usr/local/bin/python3.12 rix,
  /usr/local/bin/qurl-connector rix,
  /usr/local/bin/{app_runner,ingress_proxy,onboarding}.py r,
  /usr/local/lib/** mr,
  /app/{,**} r,

  # Runtime identity, TLS and resolver configuration.
  /etc/hosts r,
  /etc/hostname r,
  /etc/machine-id r,
  /etc/nsswitch.conf r,
  /etc/passwd r,
  /etc/group r,
  /etc/resolv.conf r,
  /etc/ssl/certs/{,**} r,
  /var/lib/dbus/machine-id r,

  # Home Assistant-owned configuration and Gateway persistence.
  /data/options.json r,
  /data/app-config.json rwk,
  /data/.app-config.json.tmp rwk,
  /data/installation-id rwk,
  /data/.installation-id.tmp rwk,
  /data/reset-connection.request rwk,
  /data/guest-activity.sqlite3{,-journal,-shm,-wal} rwk,
  /data/pages/{,**} rwk,
  /data/secrets/{,**} rwk,
  /data/connector-config/{,**} rwk,
  /data/connector-state/{,**} rwk,
  /data/logs/{,**} rwk,

  # Python, SQLite and the Connector may create bounded temporary files.
  /tmp/{,**} rwk,
  /dev/null rw,
  /dev/urandom r,

  # HTTP(S), DNS, loopback listeners and Connector transport. Network-family
  # use outside these ordinary TCP/UDP sockets will appear in the audit log.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # app_runner supervises fixed child processes in the same App container.
  signal (send, receive) set=(hup, int, kill, term, usr1, usr2, chld, cont),

  # No blanket `file`, `network` or `capability` grants.
}
